        return newSet;
      });
      setIsMultiSelectMode(true);
      return;
    }

    // Regular single click - open details
    setSelectedEvent(event);
    setPopoverAnchor(e.currentTarget as HTMLElement);
    onEventClick?.(event);
  };

  const handleSelectSlot = (slotInfo: any) => {
    // Clear multi-select when clicking empty space
    if (selectedEvents.size > 0) {
      setSelectedEvents(new Set());
      setIsMultiSelectMode(false);
    }
    onSlotClick?.({ start: slotInfo.start, end: slotInfo.end });
  };

  const handleEventDrop = ({ event, start, end }: any) => {
    // Validate: cannot move to past (with better timezone handling)
    const now = new Date();
    now.setSeconds(0, 0);
    
    if (start < now) {
      toast.error('Cannot schedule posts in the past', {
        description: 'Please select a future date and time'
      });
      return;
    }

    // Check if moving to far future (> 1 year)
    const oneYearFromNow = new Date(now);
    oneYearFromNow.setFullYear(oneYearFromNow.getFullYear() + 1);
    
    if (start > oneYearFromNow) {
      toast.warning('Scheduling more than 1 year ahead', {
        description: 'Posts scheduled far in the future may need review closer to publish time'
      });
    }

    // If multiple events selected, move all of them
    if (selectedEvents.size > 0 && selectedEvents.has(event.id)) {
      const timeDiff = start.getTime() - event.start.getTime();
      let successCount = 0;
      let failCount = 0;
      
      filteredEvents.forEach(evt => {
        if (selectedEvents.has(evt.id)) {
          const newStart = new Date(evt.start.getTime() + timeDiff);
          const newEnd = new Date(evt.end.getTime() + timeDiff);
          
          // Validate each event
          if (newStart >= now && newStart <= oneYearFromNow) {
            onEventDrop?.(evt, newStart, newEnd);
            successCount++;
          } else {
            failCount++;
          }
        }
      });

      if (successCount > 0) {
        toast.success(`Moved ${successCount} post(s) successfully`);
      }
      if (failCount > 0) {
        toast.warning(`${failCount} post(s) could not be moved (invalid date)`);
      }
      
      setSelectedEvents(new Set());
      setIsMultiSelectMode(false);
    } else {
      // Single event move
      onEventDrop?.(event, start, end);
      toast.success('Post rescheduled successfully');
    }
  };

  const handleEventResize = ({ event, start, end }: any) => {
    // Validate: cannot resize to past
    const now = new Date();
    if (start < now || end < now) {
      toast.error('Cannot schedule posts in the past');
      return;
    }

    onEventDrop?.(event, start, end);
    toast.success('Post duration updated');
  };

  const clearSelection = () => {
    setSelectedEvents(new Set());
    setIsMultiSelectMode(false);
  };

  const handleBulkDelete = () => {
    if (selectedEvents.size === 0) return;
    
    const eventIds = Array.from(selectedEvents);
    onBulkDelete?.(eventIds);
    clearSelection();
  };

    if (selectedEvents.size === 0) return;
    
    const eventIds = Array.from(selectedEvents);
    onBulkStatusChange?.(eventIds, status);

    onViewChange?.(view);
  };

  const handleNavigate = (date: Date | 'PREV' | 'NEXT' | 'TODAY') => {

      },
    };
  };

  const components = {
    event: (props: any) => (

        onRetry={onEventRetry}
        onMoveToDraft={(event) => {
          toast.info('Move to draft functionality coming soon');
        }}
        onViewOnPlatform={(event) => {
          toast.info('View on platform functionality coming soon');
        }}
      >
        <div>
          <CalendarEventCard
            {...props}
            onEdit={onEventEdit}
            onDuplicate={onEventDuplicate}
            onDelete={onEventDelete}
            onViewAnalytics={(event) => {
              toast.info('Analytics functionality coming soon');
            }}
          />

      </EventContextMenu>

    toolbar: (props: any) => (

        onProcessQueue={onProcessQueue}
        onViewQueue={onViewQueue}

    ),

                {eventCount} post{eventCount !== 1 ? 's' : ''}

  eventCount: number;
  renderTime?: number;
}

export function CalendarPerformanceMonitor({ 
  eventCount, 
  renderTime 
}: CalendarPerformanceMonitorProps) {
  const [showWarning, setShowWarning] = useState(false);

  useEffect(() => {
    // Show performance warning for large datasets
    if (eventCount > 100) {
      setShowWarning(true);
      
      // Auto-hide after 5 seconds
      const timer = setTimeout(() => setShowWarning(false), 5000);
      return () => clearTimeout(timer);
    } else {
      setShowWarning(false);
    }
  }, [eventCount]);

  // Performance thresholds
  const isHighVolume = eventCount > 100;
  const isCriticalVolume = eventCount > 500;

  if (!showWarning && eventCount <= 100) {
    return null;
  }

  return (
    <div className="space-y-2">
      {/* Performance Badge */}
      {isHighVolume && (
        <div className="flex items-center gap-2">
          <Badge 
            variant={isCriticalVolume ? "destructive" : "default"}
            className="gap-1.5"
          >
            <TrendingUp className="w-3 h-3" />
            {eventCount} events loaded
          </Badge>
          {renderTime && (
            <span className="text-xs text-muted-foreground">
              Rendered in {renderTime}ms
            </span>
          )}
        </div>
      )}

      {/* High Volume Warning */}
      {isCriticalVolume && showWarning && (
        <Alert variant="default" className="border-yellow-500/50 bg-yellow-500/5">
          <AlertTriangle className="h-4 w-4 text-yellow-600" />

  onEventDelete?: (event: CalendarEvent) => void;
  onEventRetry?: (event: CalendarEvent) => void;
  onDateClick?: (date: Date) => void;

export function CalendarWeekView({
  events,
  onEventClick,
  onEventEdit,
  onEventDuplicate,
  onEventDelete,
  onEventRetry,
  onDateClick,
}: CalendarWeekViewProps) {
  const [currentWeekStart, setCurrentWeekStart] = useState(() => 
    startOfWeek(new Date(), { weekStartsOn: 0 })
  );
  const scrollContainerRef = useRef<HTMLDivElement>(null);

  // Generate 7 days for the current week
  const weekDays = Array.from({ length: 7 }, (_, i) => addDays(currentWeekStart, i));

  const handlePrevWeek = () => {
    setCurrentWeekStart(prev => subWeeks(prev, 1));
  };

  const handleNextWeek = () => {
    setCurrentWeekStart(prev => addWeeks(prev, 1));
  };

  const handleToday = () => {
    setCurrentWeekStart(startOfWeek(new Date(), { weekStartsOn: 0 }));
  };

  // Group events by date
  const eventsByDate = (date: Date) => {
    return events.filter(event => 
      isSameDay(new Date(event.start), date)
    ).sort((a, b) => new Date(a.start).getTime() - new Date(b.start).getTime());
  };

  const currentMonth = format(currentWeekStart, 'MMMM yyyy');
  const endOfWeekMonth = format(addDays(currentWeekStart, 6), 'MMMM yyyy');
  const displayMonth = currentMonth === endOfWeekMonth 
    ? currentMonth 
    : `${format(currentWeekStart, 'MMM')} - ${endOfWeekMonth}`;

  return (
    <div className="flex flex-col h-full bg-background">
      {/* Header with navigation */}

function getSuggestion(errorMessage?: string): string | null {
  if (!errorMessage) return null;

}

import { serve } from "https://deno.land/std@0.168.0/http/server.ts";

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response(null, { headers: corsHeaders });
  }

  try {
    const { text, platform, maxLength, action } = await req.json();
    console.log(`Enhancing content for ${platform}, action: ${action}`);

    const VERTEX_AI_KEY = Deno.env.get('GOOGLE_VERTEX_AI_KEY');
    if (!VERTEX_AI_KEY) {
      throw new Error('GOOGLE_VERTEX_AI_KEY not configured');
    }

    let prompt = '';
    if (action === 'shorten') {
      prompt = `Shorten this social media post to exactly ${maxLength} characters or less while keeping the core message and impact. Maintain the tone and style. Do NOT add hashtags unless they were in the original. Return ONLY the shortened text, no explanations:\n\n${text}`;
    } else if (action === 'enhance') {
      prompt = `Improve this social media post to make it more engaging while keeping it under ${maxLength} characters. Maintain the original message. Return ONLY the enhanced text, no explanations:\n\n${text}`;
    }

    const response = await fetch(
      `https://generativelanguage.googleapis.com/v1beta/models/gemini-2.5-flash:generateContent?key=${VERTEX_AI_KEY}`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          contents: [{
            role: 'user',
            parts: [{ text: prompt }]
          }],
          generationConfig: {
            temperature: 0.7,
            maxOutputTokens: 500,
          }
        }),
      }
    );

    if (!response.ok) {
      const errorText = await response.text();
      console.error('Gemini API error:', response.status, errorText);
      throw new Error(`Gemini API error: ${response.status}`);
    }

    const data = await response.json();
    
    if (!data.candidates?.[0]?.content?.parts?.[0]?.text) {
      throw new Error('No enhanced text returned');
    }

    const enhancedText = data.candidates[0].content.parts[0].text.trim();

    return new Response(
      JSON.stringify({ enhancedText }),
      { headers: { ...corsHeaders, 'Content-Type': 'application/json' }, status: 200 }
    );
  } catch (error) {
    console.error('Error enhancing content:', error);
    return new Response(
      JSON.stringify({ error: error instanceof Error ? error.message : 'Failed to enhance content' }),
      { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
    );
  }
});

import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2';

const APP_URL = 'https://sineva-vision-forge.lovable.app';

serve(async (req) => {
  try {
    const url = new URL(req.url);
    const code = url.searchParams.get('code');
    const state = url.searchParams.get('state');
    const error = url.searchParams.get('error');
    
    if (error) {
      console.error('OAuth error:', error);
      return Response.redirect(`${APP_URL}/app/integrations?error=${error}`);
    }

    if (!code || !state) {
      throw new Error('Missing code or state parameter');
    }

    // Parse state: format is "platform:state:userId"
    const [platform, oauthState, userId] = state.split(':');

    const supabaseUrl = Deno.env.get('SUPABASE_URL')!;
    const supabaseKey = Deno.env.get('SUPABASE_SERVICE_ROLE_KEY')!;
    const supabase = createClient(supabaseUrl, supabaseKey);

    // Verify state matches what we stored
    const { data: integration, error: fetchError } = await supabase
      .from('integrations')
      .select('*')
      .eq('user_id', userId)
      .eq('platform', platform)
      .maybeSingle();

    if (fetchError || !integration) {
      throw new Error('Integration not found');
    }

    if (integration.metadata?.oauth_state !== oauthState) {
      throw new Error('Invalid state parameter');
    }

    // Exchange code for access token (platform-specific)
    const tokenEndpoints: Record<string, string> = {
      facebook: 'https://graph.facebook.com/v18.0/oauth/access_token',
      'facebook-ads': 'https://graph.facebook.com/v18.0/oauth/access_token',
      instagram: 'https://graph.facebook.com/v18.0/oauth/access_token',
      whatsapp: 'https://graph.facebook.com/v18.0/oauth/access_token',
      twitter: 'https://api.twitter.com/2/oauth2/token',
      linkedin: 'https://www.linkedin.com/oauth/v2/accessToken',
      tiktok: 'https://open-api.tiktok.com/oauth/access_token/',
      youtube: 'https://oauth2.googleapis.com/token',
      'google-ads': 'https://oauth2.googleapis.com/token',
      pinterest: 'https://api.pinterest.com/v5/oauth/token',
    };

    const tokenUrl = tokenEndpoints[platform];
    const redirectUri = `${supabaseUrl}/functions/v1/oauth-callback`;

    // Get client credentials from environment
    // Handle special cases for different platforms
    let envPrefix = platform.toUpperCase().replace('-', '_');
    let clientIdKey = 'CLIENT_ID';
    let clientSecretKey = 'CLIENT_SECRET';
    
    if (platform === 'google-ads') {
      envPrefix = 'GOOGLE';
    } else if (platform === 'youtube') {
      envPrefix = 'YOUTUBE';
    } else if (platform === 'instagram' || platform === 'whatsapp' || platform === 'facebook-ads') {
      envPrefix = 'FACEBOOK';
      clientIdKey = 'CLIENT_ID';
      clientSecretKey = 'CLIENT_SECRET';
    } else if (platform === 'facebook') {
      clientIdKey = 'CLIENT_ID';
      clientSecretKey = 'CLIENT_SECRET';
    } else if (platform === 'pinterest') {
      clientIdKey = 'APP_ID';
      clientSecretKey = 'APP_SECRET';
    } else if (platform === 'tiktok') {
      clientIdKey = 'CLIENT_KEY';
    }
    
    const clientId = Deno.env.get(`${envPrefix}_${clientIdKey}`);
    const clientSecret = Deno.env.get(`${envPrefix}_${clientSecretKey}`);

    if (!clientId || !clientSecret) {
      throw new Error(`Missing ${platform} OAuth credentials`);
    }

    // Exchange authorization code for tokens
    const tokenResponse = await fetch(tokenUrl, {
      method: 'POST',
      headers: {
        'Content-Type': 'application/x-www-form-urlencoded',
      },
      body: new URLSearchParams({
        grant_type: 'authorization_code',
        code,
        redirect_uri: redirectUri,
        client_id: clientId,
        client_secret: clientSecret,
      }),
    });

    if (!tokenResponse.ok) {
      const errorText = await tokenResponse.text();
      console.error('Token exchange failed:', errorText);
      throw new Error('Failed to exchange authorization code');
    }

    const tokens = await tokenResponse.json();

    // Calculate expiry time
    const expiresAt = tokens.expires_in 
      ? new Date(Date.now() + tokens.expires_in * 1000).toISOString()
      : null;

    // Fetch account details based on platform
    let accountDetails: any = {};
    
    try {
      if (platform === 'facebook') {
        // Get ALL Facebook Pages
        console.log('Fetching ALL Facebook pages...');
        const fbResponse = await fetch(
          `https://graph.facebook.com/v18.0/me/accounts?fields=id,name,access_token,category,tasks&access_token=${tokens.access_token}`
        );
        const fbResponseText = await fbResponse.text();
        console.log('Facebook pages response:', fbResponseText);
        
        if (fbResponse.ok) {
          const fbData = JSON.parse(fbResponseText);

import { PLATFORM_CAPABILITIES, getPlatformDisplayName } from '@/lib/social/platformValidation';

interface PlatformCharCounterProps {
  content: string;
  platform: string;
  mediaCount?: number;
}

export function PlatformCharCounter({ content, platform, mediaCount = 0 }: PlatformCharCounterProps) {
  const platformChars = getPlatformCharCount(content, platform, mediaCount);
  const maxChars = PLATFORM_CAPABILITIES[platform.toLowerCase()]?.maxTextLength || 0;
  const remaining = maxChars - platformChars;
  const percentage = (remaining / maxChars) * 100;

  const getVariant = () => {
    if (remaining < 0) return 'destructive';
    if (percentage < 10) return 'destructive';
    if (percentage < 20) return 'outline';
    return 'secondary';
  };

  const getIcon = () => {
    if (remaining < 0) {
      return <AlertCircle className="h-3 w-3" />;
    }
    return <CheckCircle2 className="h-3 w-3" />;
  };

  return (
    <div className="space-y-2">
      <Badge variant={getVariant()} className="text-xs flex items-center gap-1 w-fit">
        {getIcon()}
        <span>
          {getPlatformDisplayName(platform)}: {platformChars}/{maxChars}
          {remaining < 0 && ` (${Math.abs(remaining)} over)`}
        </span>
      </Badge>
      
      {remaining < 0 && (
        <Alert variant="destructive" className="py-2">
          <AlertCircle className="h-4 w-4" />
          <AlertDescription className="text-xs">
            Content exceeds {getPlatformDisplayName(platform)}'s character limit by {Math.abs(remaining)} characters
          </AlertDescription>
        </Alert>
      )}
    </div>
  );

import { serve } from "https://deno.land/std@0.168.0/http/server.ts";
import { createClient } from 'https://esm.sh/@supabase/supabase-js@2';

const corsHeaders = {
  'Access-Control-Allow-Origin': '*',
  'Access-Control-Allow-Headers': 'authorization, x-client-info, apikey, content-type',
};

interface QueueJob {
  id: string;
  user_id: string;
  integration_id: string;
  platform: string;
  content_id: string | null;
  post_content: string;
  media_urls: string[];
  media_ids: string[];
  scheduled_for: string;
  status: string;
  attempts: number;
  max_attempts: number;
  platform_metadata: any;
  container_id: string | null;
  publish_confirmation_needed: boolean;
}

interface Integration {
  id: string;
  platform: string;
  access_token: string;
  refresh_token: string | null;
  expires_at: string | null;
}

serve(async (req) => {
  if (req.method === 'OPTIONS') {
    return new Response(null, { headers: corsHeaders });
  }

  try {
    const supabase = createClient(
      Deno.env.get('SUPABASE_URL') ?? '',
      Deno.env.get('SUPABASE_SERVICE_ROLE_KEY') ?? ''
    );

    console.log('🤖 Posting worker started');

    // Fetch jobs that are due
    const now = new Date().toISOString();
    const { data: jobs, error: fetchError } = await supabase
      .from('posting_queue')
      .select('*')
      .in('status', ['pending', 'scheduled'])
      .lte('scheduled_for', now)
      .order('priority', { ascending: false })
      .order('scheduled_for', { ascending: true })
      .limit(10);

    if (fetchError) {
      throw new Error(`Failed to fetch jobs: ${fetchError.message}`);
    }

    if (!jobs || jobs.length === 0) {
      console.log('✅ No jobs to process');
      return new Response(
        JSON.stringify({ message: 'No jobs to process', processed: 0 }),
        { headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
      );
    }

    console.log(`📋 Found ${jobs.length} jobs to process`);
    let processed = 0;
    let succeeded = 0;
    let failed = 0;

    for (const job of jobs as QueueJob[]) {
      try {
        // Mark as processing
        await supabase
          .from('posting_queue')
          .update({ status: 'processing', last_attempt_at: new Date().toISOString() })
          .eq('id', job.id);

        // Fetch integration tokens
        const { data: integration, error: integrationError } = await supabase
          .from('integrations')
          .select('*')
          .eq('id', job.integration_id)
          .single();

        if (integrationError || !integration) {
          throw new Error(`Integration not found: ${integrationError?.message}`);
        }

        const integrationData = integration as Integration;

        // Check if token is expired
        if (integrationData.expires_at && new Date(integrationData.expires_at) < new Date()) {
          throw new Error('Access token expired. Refresh needed.');
        }

        // Process based on platform
        const result = await postToSocialMedia(job, integrationData);

        if (result.success) {
          // Mark as completed
          await supabase
            .from('posting_queue')
            .update({
              status: 'completed',
              posted_at: new Date().toISOString(),
              platform_post_id: result.postId,
              platform_response: result.response,
            })
            .eq('id', job.id);

          // Update content_calendar if linked
          if (job.content_id) {
            await supabase
              .from('content_calendar')
              .update({ published: true })
              .eq('id', job.content_id);
          }

          console.log(`✅ Job ${job.id} completed successfully`);
          succeeded++;
        } else {
          throw new Error(result.error || 'Unknown error');
        }

        processed++;
      } catch (error) {
        console.error(`❌ Job ${job.id} failed:`, error);
        
        const attempts = job.attempts + 1;
        const errorMessage = error instanceof Error ? error.message : 'Unknown error';

        if (attempts >= job.max_attempts) {
          // Max retries reached, mark as failed
          await supabase
            .from('posting_queue')
            .update({
              status: 'failed',
              attempts,
              error_message: errorMessage,
              error_details: { error: String(error) },
            })
            .eq('id', job.id);
          
          failed++;
        } else {
          // Retry with exponential backoff
          const backoffMinutes = Math.pow(2, attempts) * 5; // 5, 10, 20 minutes
          const nextAttempt = new Date(Date.now() + backoffMinutes * 60 * 1000);

          await supabase
            .from('posting_queue')
            .update({
              status: 'pending',
              attempts,
              scheduled_for: nextAttempt.toISOString(),
              error_message: errorMessage,
            })
            .eq('id', job.id);

          console.log(`🔄 Job ${job.id} scheduled for retry in ${backoffMinutes} minutes`);
        }
      }
    }

    console.log(`✅ Worker completed: ${succeeded} succeeded, ${failed} failed, ${processed} total`);

    return new Response(
      JSON.stringify({
        message: 'Worker completed',
        processed,
        succeeded,
        failed,
      }),
      { headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
    );

  } catch (error) {
    console.error('Worker error:', error);
    return new Response(
      JSON.stringify({ error: error instanceof Error ? error.message : 'Unknown error' }),
      { status: 500, headers: { ...corsHeaders, 'Content-Type': 'application/json' } }
    );
  }
});

async function postToSocialMedia(

async function postToLinkedIn(
  job: QueueJob,
  integration: Integration
): Promise<{ success: boolean; postId?: string; response?: any; error?: string }> {
  try {
    // Get LinkedIn user ID
    const linkedInUserId = await getLinkedInUserId(integration.access_token);
    if (!linkedInUserId) {
      throw new Error('Failed to fetch LinkedIn user ID');
    }

    let mediaIds = job.media_ids || [];

    // Upload media if needed
    if (job.media_urls && job.media_urls.length > 0 && mediaIds.length === 0) {
      // Register upload
      const registerResponse = await fetch(
        'https://api.linkedin.com/v2/assets?action=registerUpload',
        {
          method: 'POST',
          headers: {
            'Authorization': `Bearer ${integration.access_token}`,
            'Content-Type': 'application/json',
            'X-Restli-Protocol-Version': '2.0.0',
          },
          body: JSON.stringify({
            registerUploadRequest: {
              recipes: ['urn:li:digitalmediaRecipe:feedshare-image'],
              owner: `urn:li:person:${linkedInUserId}`,
              serviceRelationships: [{
                relationshipType: 'OWNER',
                identifier: 'urn:li:userGeneratedContent',
              }],
            },
          }),
        }
      );

      if (!registerResponse.ok) {
        throw new Error(`Media registration failed: ${await registerResponse.text()}`);
      }

      const registerData = await registerResponse.json();
      const uploadUrl = registerData.value.uploadMechanism['com.linkedin.digitalmedia.uploading.MediaUploadHttpRequest'].uploadUrl;
      const asset = registerData.value.asset;

      // Upload binary
      for (const mediaUrl of job.media_urls) {
        const mediaResponse = await fetch(mediaUrl);
        const mediaBlob = await mediaResponse.blob();

        const uploadResponse = await fetch(uploadUrl, {
          method: 'POST',
          headers: {
            'Authorization': `Bearer ${integration.access_token}`,
          },
          body: mediaBlob,
        });

        if (!uploadResponse.ok) {
          throw new Error(`Media upload failed: ${await uploadResponse.text()}`);
        }
      }

      mediaIds = [asset];
    }

    // Create post
    const postData: any = {
      author: `urn:li:person:${linkedInUserId}`,
      lifecycleState: 'PUBLISHED',
      specificContent: {
        'com.linkedin.ugc.ShareContent': {
          shareCommentary: { text: job.post_content },
          shareMediaCategory: mediaIds.length > 0 ? 'IMAGE' : 'NONE',
        },
      },
      visibility: {
        'com.linkedin.ugc.MemberNetworkVisibility': 'PUBLIC',
      },
    };

    if (mediaIds.length > 0) {
      postData.specificContent['com.linkedin.ugc.ShareContent'].media = mediaIds.map(id => ({
        status: 'READY',
        media: id,
      }));
    }

    const response = await fetch('https://api.linkedin.com/v2/ugcPosts', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${integration.access_token}`,
        'Content-Type': 'application/json',
        'X-Restli-Protocol-Version': '2.0.0',
      },
      body: JSON.stringify(postData),
    });

    if (!response.ok) {
      throw new Error(`LinkedIn post failed: ${await response.text()}`);
    }

    const result = await response.json();
    return {
      success: true,
      postId: result.id,
      response: result,
    };
  } catch (error) {
    return {
      success: false,
      error: error instanceof Error ? error.message : 'LinkedIn posting failed',
    };
  }
}

async function postToTwitter(
  job: QueueJob,
  integration: Integration
): Promise<{ success: boolean; postId?: string; response?: any; error?: string }> {
  try {
    let mediaIds = job.media_ids || [];

    // Upload media if needed
    if (job.media_urls && job.media_urls.length > 0 && mediaIds.length === 0) {
      for (const mediaUrl of job.media_urls) {
        const mediaResponse = await fetch(mediaUrl);
        const mediaBlob = await mediaResponse.blob();
        const mediaBuffer = await mediaBlob.arrayBuffer();
        const base64Media = btoa(String.fromCharCode(...new Uint8Array(mediaBuffer)));

        const formData = new FormData();
        formData.append('media_data', base64Media);

        const uploadResponse = await fetch('https://upload.twitter.com/1.1/media/upload.json', {
          method: 'POST',
          headers: {
            'Authorization': `Bearer ${integration.access_token}`,
          },
          body: formData,
        });

        if (!uploadResponse.ok) {
          throw new Error(`Media upload failed: ${await uploadResponse.text()}`);
        }

        const uploadData = await uploadResponse.json();
        mediaIds.push(uploadData.media_id_string);
      }
    }

    // Create tweet
    const tweetData: any = {
      text: job.post_content,
    };

    if (mediaIds.length > 0) {
      tweetData.media = { media_ids: mediaIds };
    }

    const response = await fetch('https://api.twitter.com/2/tweets', {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${integration.access_token}`,
        'Content-Type': 'application/json',
      },
      body: JSON.stringify(tweetData),
    });

    if (!response.ok) {
      throw new Error(`Twitter post failed: ${await response.text()}`);
    }

    const result = await response.json();
    return {
      success: true,
      postId: result.data.id,
      response: result,
    };
  } catch (error) {
    return {
      success: false,
      error: error instanceof Error ? error.message : 'Twitter posting failed',
    };
  }

async function postToYouTube(

    }

    const videoUrl = job.media_urls[0];

    // Fetch the video file
    const videoResponse = await fetch(videoUrl);
    if (!videoResponse.ok) {
      throw new Error(`Failed to fetch video: ${videoResponse.statusText}`);
    }

    const videoBlob = await videoResponse.blob();
    const videoSize = videoBlob.size;

    // Prepare metadata
    const metadata = {
      snippet: {
        title: job.post_content.slice(0, 100) || 'Untitled Video',
        description: job.post_content,
        categoryId: '22', // People & Blogs
      },
      status: {
        privacyStatus: 'public',
        selfDeclaredMadeForKids: false,
      },
    };

    // Initialize resumable upload
    const initResponse = await fetch(
      'https://www.googleapis.com/upload/youtube/v3/videos?uploadType=resumable&part=snippet,status',
      {
        method: 'POST',
        headers: {
          'Authorization': `Bearer ${integration.access_token}`,
          'Content-Type': 'application/json',
          'X-Upload-Content-Length': videoSize.toString(),
          'X-Upload-Content-Type': videoBlob.type,
        },
        body: JSON.stringify(metadata),
      }
    );

    if (!initResponse.ok) {
      throw new Error(`YouTube upload initialization failed: ${await initResponse.text()}`);
    }

    const uploadUrl = initResponse.headers.get('Location');
    if (!uploadUrl) {
      throw new Error('No upload URL received from YouTube');
    }

    // Upload video content
    const uploadResponse = await fetch(uploadUrl, {
      method: 'PUT',
      headers: {
        'Content-Type': videoBlob.type,
      },
      body: videoBlob,
    });

    if (!uploadResponse.ok) {
      throw new Error(`YouTube video upload failed: ${await uploadResponse.text()}`);
    }

    const result = await uploadResponse.json();
    return {
      success: true,
      postId: result.id,
      response: result,
    };
  } catch (error) {
    return {
      success: false,
      error: error instanceof Error ? error.message : 'YouTube upload failed',
    };
  }

): Promise<{ success: boolean; postId?: string; response?: any; error?: string }> {
  try {
    // Get Facebook Page ID from metadata
    const pageId = integration.metadata?.account_id || integration.account_id;
    const pageAccessToken = integration.metadata?.page_access_token || integration.access_token;

    if (!pageId) {
      return {
        success: false,
        error: 'Facebook Page ID not found. Please reconnect your Facebook account.',
      };
    }

    let mediaIds = job.media_ids || [];

    // Upload media if needed
    if (job.media_urls && job.media_urls.length > 0 && mediaIds.length === 0) {
      for (const url of job.media_urls) {
        const uploadResponse = await fetch(
          `https://graph.facebook.com/v18.0/${pageId}/photos`,
          {
            method: 'POST',
            headers: {
              'Content-Type': 'application/json',
            },
            body: JSON.stringify({
              url: url,
              published: false, // Upload but don't publish yet
              access_token: pageAccessToken,
            }),
          }
        );

        if (!uploadResponse.ok) {
          const error = await uploadResponse.json();
          throw new Error(`Facebook upload failed: ${error.error?.message || uploadResponse.statusText}`);
        }

        const data = await uploadResponse.json();
        mediaIds.push(data.id);
      }
    }

    // Create post
    const postData: any = {
      message: job.post_content,
      access_token: pageAccessToken,
    };

    // Add media if available
    if (mediaIds.length > 0) {
      if (mediaIds.length === 1) {
        postData.attached_media = [{ media_fbid: mediaIds[0] }];
      } else {
        postData.attached_media = mediaIds.map(id => ({ media_fbid: id }));
      }
    }

    const response = await fetch(
      `https://graph.facebook.com/v18.0/${pageId}/feed`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify(postData),
      }
    );

    if (!response.ok) {
      const error = await response.json();
      throw new Error(`Facebook post failed: ${error.error?.message || response.statusText}`);
    }

    const result = await response.json();

    return {
      success: true,
      postId: result.id,
      response: result,
    };
  } catch (error) {
    return {
      success: false,
      error: error instanceof Error ? error.message : 'Facebook posting failed',
    };
  }
}

async function postToInstagram(
  job: QueueJob,
  integration: any
): Promise<{ success: boolean; postId?: string; response?: any; error?: string }> {
  try {
    // Get Instagram Business Account ID from metadata
    const igAccountId = integration.metadata?.account_id || integration.account_id;
    const pageAccessToken = integration.metadata?.page_access_token || integration.access_token;

    if (!igAccountId) {
      return {
        success: false,
        error: 'Instagram Business Account ID not found. Please reconnect your Instagram account.',
      };
    }

    // Instagram requires media
    if (!job.media_urls || job.media_urls.length === 0) {
      return {
        success: false,
        error: 'Instagram requires at least one image or video.',
      };
    }

    const mediaUrl = job.media_urls[0];
    const isVideo = mediaUrl.match(/\.(mp4|mov|avi)$/i);

    // Step 1: Create media container
    let containerId = job.container_id;

    if (!containerId) {
      const containerData: any = {
        caption: job.post_content,
        access_token: pageAccessToken,
      };

      if (isVideo) {
        containerData.media_type = 'VIDEO';
        containerData.video_url = mediaUrl;
      } else {
        containerData.image_url = mediaUrl;
      }

      const containerResponse = await fetch(
        `https://graph.facebook.com/v18.0/${igAccountId}/media`,
        {
          method: 'POST',
          headers: {
            'Content-Type': 'application/json',
          },
          body: JSON.stringify(containerData),
        }
      );

      if (!containerResponse.ok) {
        const error = await containerResponse.json();
        throw new Error(`Instagram container creation failed: ${error.error?.message || containerResponse.statusText}`);
      }

      const containerResult = await containerResponse.json();
      containerId = containerResult.id;

      // For videos, we need to wait for processing
      if (isVideo) {
        // Check status
        let attempts = 0;
        let status = 'IN_PROGRESS';

        while (status === 'IN_PROGRESS' && attempts < 30) {
          await new Promise(resolve => setTimeout(resolve, 2000)); // Wait 2 seconds

          const statusResponse = await fetch(
            `https://graph.facebook.com/v18.0/${containerId}?fields=status_code&access_token=${pageAccessToken}`
          );

          if (statusResponse.ok) {
            const statusData = await statusResponse.json();
            status = statusData.status_code || 'FINISHED';
          }

          attempts++;
        }

        if (status !== 'FINISHED') {
          return {
            success: false,
            error: `Video processing timeout. Status: ${status}. Please try again later.`,
          };
        }
      }
    }

    // Step 2: Publish the container
    const publishResponse = await fetch(
      `https://graph.facebook.com/v18.0/${igAccountId}/media_publish`,
      {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          creation_id: containerId,
          access_token: pageAccessToken,
        }),
      }
    );

    if (!publishResponse.ok) {
      const error = await publishResponse.json();
      throw new Error(`Instagram publish failed: ${error.error?.message || publishResponse.statusText}`);
    }

    const result = await publishResponse.json();

    return {
      success: true,
      postId: result.id,
      response: result,
    };
  } catch (error) {
    return {
      success: false,
      error: error instanceof Error ? error.message : 'Instagram posting failed',
    };
  }
}

import { useEffect, useState } from 'react';
import { supabase } from '@/integrations/supabase/client';

export interface ConnectedIntegration {
  id: string;
  platform: string;
  accountName?: string;
  accountId?: string;
  connected: boolean;
  lastSync?: Date;
  expiresAt?: string | null;
  metadata?: any;
}

export function useConnectedIntegrations() {
  const [integrations, setIntegrations] = useState<ConnectedIntegration[]>([]);
  const [loading, setLoading] = useState(true);

  const fetchIntegrations = async () => {
    try {
      const { data: { user } } = await supabase.auth.getUser();
      if (!user) {
        setIntegrations([]);
        return;
      }

      const { data, error } = await supabase
        .from('integrations')
        .select('*')
        .eq('user_id', user.id)
        .order('created_at', { ascending: false });

      if (error) throw error;

      const connectedIntegrations: ConnectedIntegration[] = (data || []).map((integration: any) => ({
        id: integration.id,
        platform: integration.platform,
        accountName: integration.metadata?.account_name || integration.account_name || undefined,
        accountId: integration.metadata?.account_id || integration.account_id || undefined,
        connected: !!integration.access_token,
        lastSync: integration.updated_at ? new Date(integration.updated_at) : undefined,
        expiresAt: integration.expires_at || null,
        metadata: integration.metadata || {},
      }));

      setIntegrations(connectedIntegrations);
    } catch (error) {
      console.error('Error fetching integrations:', error);
      setIntegrations([]);
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    fetchIntegrations();

    // Subscribe to changes in integrations table
    const channel = supabase
      .channel('integrations-changes')
      .on(
        'postgres_changes',
        {
          event: '*',
          schema: 'public',
          table: 'integrations',
        },
        () => {
          fetchIntegrations();
        }
      )
      .subscribe();

    return () => {
      supabase.removeChannel(channel);
    };
  }, []);

  const isConnected = (platform: string): boolean => {
    return integrations.some(
      (int) => int.platform.toLowerCase() === platform.toLowerCase() && int.connected
    );
  };

  const getConnectedIntegration = (platform: string): ConnectedIntegration | undefined => {
    return integrations.find(
      (int) => int.platform.toLowerCase() === platform.toLowerCase() && int.connected
    );
  };

  const getConnectedPlatforms = (): string[] => {
    return integrations
      .filter((int) => int.connected)
      .map((int) => int.platform);
  };

  return {
    integrations,
    loading,
    isConnected,
    getConnectedIntegration,
    getConnectedPlatforms,
    refetch: fetchIntegrations,
  };
}
